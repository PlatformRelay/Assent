```python
#!/usr/bin/env python3
# compare cli.md:98-251 at the pre-task sha, with the two Goal-named navigation-only
# transformations applied, against the five sections of the new operating-safely page.
# Expect: no output (byte-identical after normalisation).
import re, subprocess, sys

OLD_SHA = "0c83e61"
old = subprocess.run(["git", "show", f"{OLD_SHA}:docs/usage/cli.md"],
                     capture_output=True, text=True, check=True).stdout.splitlines()[97:251]
new = open("docs/usage/operating-safely.md").read().splitlines()
start = next(i for i, l in enumerate(new) if l.startswith("## What gates approve and merge"))
new = new[start:]

def norm(lines):
    out = []
    for l in lines:
        l = re.sub(r"^### ", "## ", l)                                   # heading level
        l = l.replace("](#assent-doctor)", "](cli.md#assent-doctor)")    # doctor anchor
        l = l.replace("reruns the CI snippet above — which passes no `--pack`",
                      "reruns the [`assent run` invocation](cli.md#assent-run) — which passes no `--pack`")
        out.append(l)
    while out and not out[-1].strip(): out.pop()                          # trailing blanks
    return out

old_n, new_n = norm(old), norm(new)
for i, (a, b) in enumerate(zip(old_n, new_n)):
    if a != b:
        print(f"DIFF line {i+1}:\n  old: {a!r}\n  new: {b!r}"); sys.exit(1)
if len(old_n) != len(new_n):
    print(f"LENGTH {len(old_n)} != {len(new_n)}"); sys.exit(1)
print("fidelity: identical after normalisation")
```
